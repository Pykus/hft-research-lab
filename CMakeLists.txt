cmake_minimum_required(VERSION 3.16)
project(hft_research_lab VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(hft_core src/order.cpp)
target_include_directories(hft_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(hft_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(hft_demo src/main.cpp)
target_link_libraries(hft_demo PRIVATE hft_core)
enable_testing()
add_executable(hft_tests tests/order_test.cpp)
target_link_libraries(hft_tests PRIVATE hft_core)
add_test(NAME order_validation COMMAND hft_tests)
